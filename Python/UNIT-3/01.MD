def count_digits(number):
    number = abs(number)

    count = 0

    if number == 0:
        return 1

    while number > 0:
        count = count + 1
        number = number // 10

    return count

count = count_digits(4826)
print("Total digits:", count)

    number = abs(number)
    reverse = 0

    while number > 0:
        digit = number % 10
        reverse = reverse * 10 + digit
        number = number // 10

    return reverse

def is_palindrome(number):
    original = number
    number = abs(number)
    reverse = 0

    while number > 0:
        digit = number % 10
        reverse = reverse * 10 + digit
        number = number // 10

    return original >= 0 and original == reverse

result = is_palindrome(121)
print(result)

def is_prime(number):
    if number <= 1:
        return False

    i = 2

    while i < number:
        if number % i == 0:
            return False

        i = i + 1

    return True

result = is_prime(17)
print(result)

def display_primes(start, end):
    count = 0

    print("Prime numbers:", end=" ")

    for number in range(start, end + 1):
        if number > 1:
            is_prime = True
            i = 2

            while i < number:
                if number % i == 0:
                    is_prime = False
                    break

                i = i + 1

            if is_prime:
                print(number, end=" ")
                count = count + 1

    if count == 0:
        print("No prime numbers", end="")

    print()
    print("Total prime numbers:", count)

display_primes(10, 30)

def display_student_report(name, mark_1, mark_2, mark_3):
    total = mark_1 + mark_2 + mark_3
    percentage = (total / 300) * 100

    if percentage >= 40:
        result = "PASS"
    else:
        result = "FAIL"

    print("===============================")
    print("        STUDENT REPORT")
    print("===============================")
    print("Name       :", name)
    print("Subject 1  :", mark_1)
    print("Subject 2  :", mark_2)
    print("Subject 3  :", mark_3)
    print("--------------------------------")
    print("Total      :", total, "/ 300")
    print("Percentage :", round(percentage, 2), "%")
    print("Result     :", result)
    print("===============================")

display_student_report("Rahul", 75, 80, 65)
